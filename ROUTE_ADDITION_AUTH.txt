/**
 * @swagger
 * /login:
 *   post:
 *     summary: Unified login for both users and companies (auto-detects account type)
 *     tags: [Auth]
 *     requestBody:
 *       required: true
 *       content:
 *         application/json:
 *           schema:
 *             type: object
 *             required:
 *               - email
 *               - password
 *             properties:
 *               email:
 *                 type: string
 *               password:
 *                 type: string
 *     responses:
 *       200:
 *         description: Login successful - returns full user/company data and session info
 *       401:
 *         description: Invalid credentials or account not found
 */
router.post(
  '/login',
  [
    body('email').isEmail().withMessage('Valid email required'),
    body('password').notEmpty().withMessage('Password required')
  ],
  validate,
  authController.loginUnified  // Add this function to auth.controller.js
);
